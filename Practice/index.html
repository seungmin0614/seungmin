MicroPython ESP32 펌웨어 가이드 (개인 라이브러리 연동)

제출해주신 Thonny 코드를 기반으로 보완된 MicroPython 소스코드입니다. ble_library.py와 연동하여 RGB LED (빨강 GPIO 25, 초록 GPIO 26, 파랑 GPIO 27)를 모델 분류에 따라 실시간 색상 제어합니다. 

# ESP32 MicroPython RGB LED BLE Controller (user custom library)
from machine import Pin
from time import sleep
import ble_library
import bluetooth

# BLE 초기화 및 기기 검색용 명칭 등록 (웹 앱에서 필터링 가능)
ble = bluetooth.BLE()
p = ble_library.BLESimplePeripheral(ble, "ESP_min") 

# RGB LED 각 핀 지정 (Red: 25, Green: 26, Blue: 27)
R = Pin(25, Pin.OUT)
G = Pin(26, Pin.OUT)
B = Pin(27, Pin.OUT)

# 초기 상태: 모두 끄기
R.off()
G.off()
B.off()

# 수신된 BLE 데이터에 따라 RGB 색상 제어하는 콜백 함수
def on_rx(v): 
    try:
        # 데이터가 bytes로 수신되므로 문자열로 디코딩하고 공백/줄바꿈을 제거합니다.
        msg = v.decode("utf-8").strip()
        print("Received Signal:", msg)
        
        # '1' : 빨간색 켬
        if msg == '1':
            R.on()
            G.off()
            B.off()
            
        # '2' : 초록색 켬
        elif msg == '2':
            R.off()
            G.on()
            B.off()
            
        # '3' : 파란색 켬
        elif msg == '3':
            R.off()
            G.off()
            B.on()
            
        # '4' : 모든 LED 켬 (흰색)
        elif msg == '4':
            R.on()
            G.on()
            B.on()
            
        # '5' : 모든 LED 끔 (Black)
        elif msg == '5':
            R.off()
            G.off()
            B.off()
            
    except Exception as e:
        print("Data decode error:", e)

# 데이터가 전송될 때마다 전등색 변경 함수 호출
p.on_write(on_rx)
print("[SYSTEM] BLESimplePeripheral is ready. Device Name: ESP_min")
